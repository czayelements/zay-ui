Select

A custom select with keyboard typeahead and a proper listbox. Pass `items` so the trigger can show the label.

const fruits = [
  { label: "Apple", value: "apple" },
  { label: "Pear", value: "pear" },
  { label: "Plum", value: "plum" },
];

<Select items={fruits}>
  <SelectTrigger placeholder="Pick a fruit" className="w-56" />
  <SelectContent>
    {fruits.map((f) => <SelectItem key={f.value} value={f.value}>{f.label}</SelectItem>)}
  </SelectContent>
</Select>
